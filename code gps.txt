void oled_init(void) { 
GpioDataRegs.GPBCLEAR.bit.GPIO61 = 1; 
writecmd(0x38); //function set (DL 8bit, English) 
DELAY_US(2000); 
writecmd(0x08); 
//display off 
DELAY_US(1000); 
writecmd(0x01); 
//display clear 
DELAY_US (3000); 
writecmd(0x06); 
//entry mode set (cursor auto increment) 
DELAY_US(1000); 
//home command (cursor initial position) 
writecmd(0x02); 
DELAY_US(1000); 
writecmd(0x0F); //display on 
DELAY_US(1000); 
GpioDataRegs.GPBSET.bit.GPIO61=1; //SET CS 
EDIS; 
} 
void writecmd(char cmd) 
{ 
EALLOW; 
GpioDataRegs.GPBCLEAR.bit.GP1061=1; //clear CS 
GpioDataRegs, GPBCLEAR.bit.GPIO611; //clear CS 
while(SpiaRegs.SPISTS.bit、BUFFULL_FLAG == 1); 
SpiaRegs.SPITXBUF ((OLED_CMD <<8) | cmd) << 6; 
DELAY_US (1000); 
GpioDataRegs.GPBSET.bit.GPIO61= 1; //SET CS 
EDIS; 
}
